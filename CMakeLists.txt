cmake_minimum_required(VERSION 3.22)
project(RecLox VERSION 0.1.0 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "Rec-lox currently supports Windows 10 and Windows 11 only.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(APP_ICON_PATH "${CMAKE_CURRENT_SOURCE_DIR}/assets/rec-lox.ico")
configure_file(src/app.rc.in "${CMAKE_CURRENT_BINARY_DIR}/app.rc" @ONLY)
add_executable(Rec-lox WIN32 src/main.cpp "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
target_include_directories(Rec-lox PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_compile_options(Rec-lox PRIVATE /W4 /permissive- /EHsc /utf-8)
target_link_libraries(Rec-lox PRIVATE mfplat mfreadwrite mfuuid ole32 shell32 uuid user32 gdi32)
set_target_properties(Rec-lox PROPERTIES OUTPUT_NAME "Rec-lox" WIN32_EXECUTABLE TRUE)
